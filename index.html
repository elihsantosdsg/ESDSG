    Google Drive

    Envia uma cópia completa para a sua conta Google e atualiza as abas da planilha (clientes, tabela de preço, formas de pagamento e documentos). A conexão é feita por um Apps Script que você cria na própria planilha. O passo a passo está mais abaixo.

    Neste dispositivo

    Seus dados ficam salvos só neste navegador. Baixe uma cópia em arquivo de vez em quando.

    Como conectar a uma planilha do Google
    1. Crie uma planilha no Google Drive (pode chamar “ESDSG Backup”).
    2. Na planilha, abra Extensões › Apps Script, apague o código que estiver lá e cole o código abaixo.
    3. Troque TROQUE-ESTA-SENHA por uma senha só sua e clique em Salvar.
    4. Clique em Implantar › Nova implantação › App da Web. Em “Executar como” deixe Eu e em “Quem tem acesso” escolha Qualquer pessoa. Autorize quando o Google pedir.
    5. Copie o URL do app da Web, cole no campo acima junto com a senha e clique em Testar conexão.
    6. Se mudar o código depois, use Implantar › Gerenciar implantações e publique uma nova versão.
    // ESDSG Financeiro · Backup (cole em Extensões > Apps Script da planilha)
const SENHA = 'TROQUE-ESTA-SENHA';
const ARQUIVO = 'esdsg-backup.json';

function doGet() {
  return ContentService.createTextOutput('ESDSG Backup ativo');
}

function doPost(e) {
  try {
    const d = JSON.parse(e.postData.contents);
    if (d.senha !== SENHA) return resp({ ok: false, erro: 'Senha incorreta.' });
    if (d.acao === 'teste') return resp({ ok: true });
    if (d.acao === 'ler') {
      const it = DriveApp.getFilesByName(ARQUIVO);
      if (!it.hasNext()) return resp({ ok: false, erro: 'Nenhum backup encontrado no Drive.' });
      return resp({ ok: true, backup: JSON.parse(it.next().getBlob().getDataAsString()) });
    }
    const txt = JSON.stringify(d.backup);
    const it = DriveApp.getFilesByName(ARQUIVO);
    if (it.hasNext()) it.next().setContent(txt);
    else DriveApp.createFile(ARQUIVO, txt, MimeType.PLAIN_TEXT);
    atualizarAbas(d.backup, txt.length);
    return resp({ ok: true });
  } catch (err) {
    return resp({ ok: false, erro: String(err) });
  }
}

function resp(o) {
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}

function atualizarAbas(b, tam) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const lista = k => { try { return JSON.parse(b.dados[k] || '[]') || []; } catch (e) { return []; } };
  const tipos = { orc: 'Orçamento', rec: 'Recibo', pgt: 'Infos de pagamento' };
  escreveAba(ss, 'Clientes', ['Nome', 'CPF/CNPJ', 'Telefone', 'E-mail'],
    lista('clientes').map(c => [c.nome, c.doc || '', c.tel || '', c.email || '']));
  escreveAba(ss, 'Tabela de preço', ['Item', 'Valor', 'Descrição'],
    lista('artes').map(a => [a.nome, a.valor, a.desc || '']));
  escreveAba(ss, 'Formas de pagamento', ['Nome', 'Chave Pix', 'Titular', 'Instituição', 'Detalhes'],
    lista('formas').map(f => [f.nome, f.chave || '', f.cnome || '', f.cinst || '', f.detalhes || '']));
  escreveAba(ss, 'Documentos', ['Tipo', 'Número', 'Cliente', 'Valor', 'Status', 'Data'],
    lista('historico').map(h => [tipos[h.t] || h.t, h.n, h.c || '', h.v || 0, h.st || '', h.d]));
  const log = ss.getSheetByName('Backups') || ss.insertSheet('Backups');
  if (log.getLastRow() === 0) log.appendRow(['Data', 'Tamanho (caracteres)']);
  log.appendRow([new Date(), tam]);
}

function escreveAba(ss, nome, cab, linhas) {
  const sh = ss.getSheetByName(nome) || ss.insertSheet(nome);
  sh.clear();
  sh.getRange(1, 1, 1, cab.length).setValues([cab]).setFontWeight('bold');
  if (linhas.length) sh.getRange(2, 1, linhas.length, cab.length).setValues(linhas);
  sh.setFrozenRows(1);
}

    O arquivo completo (com as prévias dos documentos) fica no seu Drive como esdsg-backup.json. Ele tem dados de clientes e CPF/CNPJ: não compartilhe a planilha nem o endereço do app.
